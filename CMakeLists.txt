cmake_minimum_required(VERSION 3.25)
project(FasterShadows VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FPS_BUILD_PLUGIN "Build the Windows F4SE plugin" ON)
include(CTest)
if(BUILD_TESTING)
    add_executable(ShadowPolicyTests tests/ShadowPolicyTests.cpp)
    target_include_directories(ShadowPolicyTests PRIVATE src)
    add_test(NAME ShadowPolicy COMMAND ShadowPolicyTests)
    add_executable(ShadowOptimizationTests tests/ShadowOptimizationTests.cpp)
    target_include_directories(ShadowOptimizationTests PRIVATE src)
    add_test(NAME ShadowOptimizations COMMAND ShadowOptimizationTests)
    add_executable(ShadowCpuBenchmark tests/ShadowCpuBenchmark.cpp)
    target_include_directories(ShadowCpuBenchmark PRIVATE src)
endif()
if(FPS_BUILD_PLUGIN)
    if(NOT WIN32)
        message(FATAL_ERROR "The F4SE plugin requires Windows. Set FPS_BUILD_PLUGIN=OFF for policy tests.")
    endif()
    set(ENABLE_FALLOUT_F4 ON CACHE BOOL "" FORCE)
    set(ENABLE_FALLOUT_NG ON CACHE BOOL "" FORCE)
    set(ENABLE_FALLOUT_VR ON CACHE BOOL "" FORCE)
    add_subdirectory(external/CommonLibF4/CommonLibF4 EXCLUDE_FROM_ALL)
    add_library(FasterShadows SHARED src/main.cpp src/Settings.cpp src/MenuSettings.cpp src/ShadowHooks.cpp)
    target_include_directories(FasterShadows PRIVATE src)
    target_link_libraries(FasterShadows PRIVATE CommonLibF4::CommonLibF4)
    target_compile_definitions(FasterShadows PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_precompile_headers(FasterShadows PRIVATE src/PCH.h)
    target_compile_options(FasterShadows PRIVATE /W4 /permissive- /utf-8 /EHsc /Zc:preprocessor /Zi)
    target_link_options(FasterShadows PRIVATE /DEBUG /PDBALTPATH:%_PDB%)
    install(TARGETS FasterShadows RUNTIME DESTINATION F4SE/Plugins)
    install(FILES package/FasterShadows.ini DESTINATION F4SE/Plugins)
    install(FILES LICENSE EXCEPTIONS.md THIRD_PARTY.md README.md DESTINATION docs/FasterShadows)
endif()
