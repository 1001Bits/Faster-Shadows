cmake_minimum_required(VERSION 3.25)
project(FPSPrivateBenchmark LANGUAGES CXX ASM_MASM)
set(CMAKE_CXX_STANDARD 23)
get_filename_component(FPS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SUPPORT "${FPS_ROOT}/.runtime-tests/support")
set(DEPS "C:/Development/Texture Downscaler FO4/build/vcpkg_installed/x64-windows-static-md" CACHE PATH "Existing dependencies")
find_package(spdlog CONFIG REQUIRED PATHS "${DEPS}/share/spdlog" NO_DEFAULT_PATH)
find_package(mmio CONFIG REQUIRED PATHS "${DEPS}/share/mmio" NO_DEFAULT_PATH)
add_library(detours STATIC "${SUPPORT}/Detours/src/detours.cpp" "${SUPPORT}/Detours/src/modules.cpp"
    "${SUPPORT}/Detours/src/disasm.cpp" "${SUPPORT}/Detours/src/image.cpp" "${SUPPORT}/Detours/src/creatwth.cpp")
target_include_directories(detours PUBLIC "${SUPPORT}/Detours/src")
target_compile_definitions(detours PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
add_library(FPSFolderGuard SHARED FolderGuard.cpp FolderGuard.def)
target_link_libraries(FPSFolderGuard PRIVATE detours shell32 ole32)
add_executable(DTLBenchLoader WIN32 "${SUPPORT}/private-loader/main.cpp" "${SUPPORT}/private-loader/remote.asm")
target_compile_definitions(DTLBenchLoader PRIVATE UNICODE _UNICODE)
target_link_libraries(DTLBenchLoader PRIVATE user32)
add_library(FPSBenchProbe SHARED "${SUPPORT}/probe-fo4/Probe.cpp")
set_target_properties(FPSBenchProbe PROPERTIES OUTPUT_NAME DTLBenchFO4)
add_library(FPSBenchmarkCandidate SHARED "${FPS_ROOT}/src/main.cpp" "${FPS_ROOT}/src/Settings.cpp" "${FPS_ROOT}/src/MenuSettings.cpp" "${FPS_ROOT}/src/ShadowHooks.cpp")
set_target_properties(FPSBenchmarkCandidate PROPERTIES OUTPUT_NAME FasterShadows)
foreach(target FPSBenchProbe FPSBenchmarkCandidate)
    target_include_directories(${target} PRIVATE "${FPS_ROOT}/src" "${FPS_ROOT}/external/CommonLibF4/CommonLibF4/include")
    target_compile_definitions(${target} PRIVATE ENABLE_FALLOUT_F4=1 ENABLE_FALLOUT_NG=1 WIN32_LEAN_AND_MEAN NOMINMAX)
    target_precompile_headers(${target} PRIVATE "${FPS_ROOT}/src/PCH.h")
    target_compile_options(${target} PRIVATE /EHsc /O2 /utf-8 /permissive- /Zc:preprocessor /wd4324 /wd4200 /wd4201)
    target_link_libraries(${target} PRIVATE "${FPS_ROOT}/build-release/external/CommonLibF4/CommonLibF4/Release/CommonLibF4.lib"
        spdlog::spdlog mmio::mmio advapi32 bcrypt d3d11 d3dcompiler dbghelp dxgi ole32 shell32 version)
endforeach()
target_compile_definitions(FPSBenchmarkCandidate PRIVATE FPS_ENABLE_PRIVATE_BENCHMARK=1)
